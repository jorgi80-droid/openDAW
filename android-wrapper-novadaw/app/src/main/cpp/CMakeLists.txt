cmake_minimum_required(VERSION 3.22.1)
project(studiowave_oboe LANGUAGES CXX)
find_package(oboe REQUIRED CONFIG)
add_library(studiowave_audio SHARED native-audio.cpp)
target_compile_features(studiowave_audio PRIVATE cxx_std_17)
target_compile_options(studiowave_audio PRIVATE -Wall -Wextra -Werror=return-type)
target_link_libraries(studiowave_audio oboe::oboe log)
